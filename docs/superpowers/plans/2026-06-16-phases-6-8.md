# RetrospectIQ Phases 6–8 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a full PDF-to-retrofit-intelligence pipeline: extract text from permit PDFs, analyze with OpenAI, score with a rule-based engine, and display real results on the dashboard.

**Architecture:** A "Run AI Retrofit Analysis" Client Component button calls `POST /api/permits/[id]/process`, which runs extract → analyze → score in sequence and persists results to MongoDB. The dashboard page stays a Server Component and re-renders via `router.refresh()` after the button completes.

**Tech Stack:** Next.js 16 App Router, TypeScript, MongoDB (native driver), AWS S3, `pdf-parse`, OpenAI SDK (`gpt-4o-mini`), Vitest for scoring unit tests.

## Global Constraints

- All routes are App Router Route Handlers (`app/api/.../route.ts`) — no Pages Router
- `params` in route handlers is `Promise<{ id: string }>` — always `await params`
- No Server Actions — button uses `fetch()` + `router.refresh()`
- `lib/` modules are thin wrappers around DB/external calls; business logic lives in lib, routes are thin
- Typed result objects for expected states (e.g. `scanned_pdf`); throw only for unexpected server errors
- Every run is a full overwrite — no partial updates to `building_profiles` or `predictions`
- All estimates labeled as prototype — not certified energy audits
- `OPENAI_API_KEY` must be present in `.env.local`
- `serverExternalPackages: ['pdf-parse']` required in `next.config.ts`

```ts
  id: string;
  buildingName: string;
  address: string;
  s3Key: string;
  originalFilename: string;
  sizeBytes: number;
  contentType: string;
  uploadedAt: string;
  status: PermitStatus;
  rawText?: string;

    status: "completed" | "failed" | "scanned_pdf";
    characterCount: number;
    extractedAt: string;
    error?: string;

  return {
    id: p._id.toString(),
    buildingName: p.buildingName,
    address: p.address,
    s3Key: p.s3Key,
    originalFilename: p.originalFilename,
    sizeBytes: p.sizeBytes,
    contentType: p.contentType,
    uploadedAt: p.uploadedAt.toISOString(),
    status: p.status,
    rawText: p.rawText,
    textExtraction: p.textExtraction
      ? {
          status: p.textExtraction.status,
          characterCount: p.textExtraction.characterCount,
          extractedAt: p.textExtraction.extractedAt.toISOString(),
          error: p.textExtraction.error,
        }
      : undefined,
  };
}

async function permitsCollection(): Promise<Collection<Permit>> {
  const db = await getDb();
  return db.collection<Permit>("permits");
}

export async function createPermit(input: {
  buildingName: string;
  address: string;
  s3Key: string;
  originalFilename: string;
  sizeBytes: number;
  contentType: string;
}): Promise<string> {
  const col = await permitsCollection();
  const result = await col.insertOne({
    ...input,
    uploadedAt: new Date(),
    status: "uploaded",
  });
  return result.insertedId.toString();
}

export async function getPermit(id: string): Promise<PermitView | null> {
  if (!ObjectId.isValid(id)) return null;
  const col = await permitsCollection();
  const doc = await col.findOne({ _id: new ObjectId(id) });
  return doc ? toView(doc) : null;
}

export async function listPermits(): Promise<PermitView[]> {
  const col = await permitsCollection();
  const docs = await col.find().sort({ uploadedAt: -1 }).toArray();
  return docs.map(toView);
}

export async function updatePermit(
  id: string,
  fields: Partial<Pick<Permit, "status" | "rawText" | "textExtraction">>
): Promise<void> {
  if (!ObjectId.isValid(id)) throw new Error(`Invalid permit id: ${id}`);
  const col = await permitsCollection();
  await col.updateOne({ _id: new ObjectId(id) }, { $set: fields });
}
```

### Task 3: Scoring algorithm (TDD)

**Files:**
- Create: `lib/scoring.test.ts`
- Create: `lib/scoring.ts`

**Interfaces:**
- Consumes: `BuildingProfile` from `./building-profiles`, `Prediction` / `Recommendation` from `./predictions`
- Produces: `computeRetrofitScore(profile: BuildingProfile): Omit<Prediction, 'permitId' | 'generatedAt'>`

**Scoring rules summary:**
- Base from age: `<1970→65`, `1970-84→55`, `1985-99→40`, `2000-14→25`, `≥2015→10`, unknown→40
- HVAC: old RTU/packaged unit→+15, boiler/vintage chiller→+10, heat pump/VRF/mini-split→-10
- Lighting: fluorescent/incandescent/HID→+10, LED→-15
- Controls penalty: +10 if no Controls/Automation upgrade detected AND hvac adjustment ≤ 0 (avoids double-penalizing old HVAC)
- Recent upgrades: -8 per upgrade with year ≥ 2015, capped at -30
- Clamp result to 0–100
- Priority: 0-30→Low, 31-60→Moderate, 61-80→High, 81-100→Critical
- Estimates: energyReduction = `score * 0.4`, savings = `sqft * 1.2 * (score/100)` or `score * 400`, carbon = `savings / 180`

```ts
import type { BuildingProfile } from './building-profiles'
import type { Prediction, Recommendation } from './predictions'

function ageBase(year: number | null): number {
  if (year === null) return 40
  if (year >= 2015) return 10
  if (year >= 2000) return 25
  if (year >= 1985) return 40
  if (year >= 1970) return 55
  return 65
}

function hvacAdj(hvac: string | null): number {
  if (!hvac) return 0
  const s = hvac.toLowerCase()
  if (s.includes('heat pump') || s.includes('vrf') || s.includes('mini-split')) return -10
  if (
    s.includes('rtu') ||
    s.includes('rooftop package') ||
    s.includes('packaged unit') ||
    s.includes('packaged rooftop')
  )
    return 15
  if (s.includes('boiler') || s.includes('chiller')) return 10
  return 0
}

function lightingAdj(lighting: string | null): number {
  if (!lighting) return 0
  const s = lighting.toLowerCase()
  if (s.includes('led')) return -15
  if (s.includes('fluorescent') || s.includes('incandescent') || s.includes('hid')) return 10
  return 0
}

function hasControlsUpgrade(upgrades: BuildingProfile['detectedUpgrades']): boolean {
  return upgrades.some(u => u.type === 'Controls' || u.type === 'Automation')
}

function upgradeReduction(upgrades: BuildingProfile['detectedUpgrades']): number {
  const recent = upgrades.filter(u => u.year !== null && u.year >= 2015).length
  return Math.min(recent * 8, 30)
}

function priorityLabel(score: number): Prediction['priority'] {
  if (score <= 30) return 'Low'
  if (score <= 60) return 'Moderate'
  if (score <= 80) return 'High'
  return 'Critical'
}

function buildRecommendations(profile: BuildingProfile): Recommendation[] {
  const recs: Recommendation[] = []

  if (profile.lightingType) {
    const s = profile.lightingType.toLowerCase()
    if (s.includes('fluorescent') || s.includes('incandescent') || s.includes('hid')) {
      recs.push({
        name: 'LED Lighting Conversion',
        category: 'Lighting',
        priority: 'High',
        reason: `Current lighting (${profile.lightingType}) is significantly less efficient than LED.`,
      })
    }
  }

  if (profile.hvacSystem) {
    const s = profile.hvacSystem.toLowerCase()
    if (
      s.includes('rtu') ||
      s.includes('rooftop package') ||
      s.includes('packaged unit') ||
      s.includes('packaged rooftop') ||
      s.includes('boiler')
    ) {
      recs.push({
        name: 'High-Efficiency HVAC Upgrade',
        category: 'HVAC',
        priority: 'High',
        reason: `Current system (${profile.hvacSystem}) is a strong candidate for high-efficiency replacement.`,
      })
    }
  }

  if (!hasControlsUpgrade(profile.detectedUpgrades)) {
    recs.push({
      name: 'Smart Occupancy Controls',
      category: 'Controls',
      priority: 'Medium',
      reason: 'No controls upgrade detected. Occupancy-based controls typically reduce energy use 15–25%.',
    })
    recs.push({
      name: 'Building Automation System',
      category: 'Automation',
      priority: 'Medium',
      reason:
        'No building automation detected. A BAS integrates HVAC, lighting, and controls for optimized energy management.',
    })
  }

  const nullCount = [
    profile.buildingYear,
    profile.buildingType,
    profile.squareFootage,
    profile.hvacSystem,
    profile.lightingType,
    profile.electricalService,
  ].filter(v => v === null).length

  if (nullCount >= 3) {
    recs.push({
      name: 'Professional Energy Audit',
      category: 'Audit',
      priority: 'High',
      reason:
        'Insufficient data in permit to fully assess retrofit potential. A professional audit would provide accurate baselines.',
    })
  }

  return recs
}

export function computeRetrofitScore(
  profile: BuildingProfile
): Omit<Prediction, 'permitId' | 'generatedAt'> {
  let score = ageBase(profile.buildingYear)
  const hAdj = hvacAdj(profile.hvacSystem)
  score += hAdj
  score += lightingAdj(profile.lightingType)

  // Controls penalty applied independently; skip if HVAC already contributes a positive penalty
  // (prevents double-penalizing old HVAC systems whose controls status is unknown)
  if (!hasControlsUpgrade(profile.detectedUpgrades) && hAdj <= 0) {
    score += 10
  }

  score -= upgradeReduction(profile.detectedUpgrades)
  score = Math.max(0, Math.min(100, score))

  const sqft = profile.squareFootage
  const estimatedAnnualSavings = sqft
    ? Math.round((sqft * 1.2 * (score / 100)) / 100) * 100
    : score * 400

  return {
    retrofitImpactScore: score,
    priority: priorityLabel(score),
    estimatedEnergyReductionPercent: Math.round(score * 0.4),
    estimatedAnnualSavings,
    estimatedCarbonReductionTons: Math.round(estimatedAnnualSavings / 180),
    recommendations: buildRecommendations(profile),
    modelType: 'prototype_rule_based',
  }
}
```

```ts
import pdfParse from 'pdf-parse'
import { GetObjectCommand } from '@aws-sdk/client-s3'
import { getS3Client, getS3Bucket } from './s3'
import { getPermit, updatePermit } from './permits'

export type ExtractionResult = {
  outcome: 'completed' | 'scanned_pdf' | 'failed'
  error?: string
}

const MIN_CHARS = 100
const MAX_CHARS = 100_000

async function downloadPdf(s3Key: string): Promise<Buffer> {
  const cmd = new GetObjectCommand({ Bucket: getS3Bucket(), Key: s3Key })
  const res = await getS3Client().send(cmd)
  const stream = res.Body as AsyncIterable<Uint8Array>
  const chunks: Buffer[] = []
  for await (const chunk of stream) {
    chunks.push(Buffer.from(chunk))
  }
  return Buffer.concat(chunks)
}

export async function extractPermitText(permitId: string): Promise<ExtractionResult> {
  const permit = await getPermit(permitId)
  if (!permit) throw new Error(`Permit not found: ${permitId}`)

  const buffer = await downloadPdf(permit.s3Key)
  const parsed = await pdfParse(buffer)
  const rawText = parsed.text.slice(0, MAX_CHARS)
  const characterCount = rawText.replace(/\s/g, '').length

  if (characterCount < MIN_CHARS) {
    await updatePermit(permitId, {
      status: 'scanned_pdf',
      rawText: '',
      textExtraction: { status: 'scanned_pdf', characterCount: 0, extractedAt: new Date() },
    })
    return { outcome: 'scanned_pdf' }
  }

  await updatePermit(permitId, {
    status: 'text_extracted',
    rawText,
    textExtraction: { status: 'completed', characterCount, extractedAt: new Date() },
  })
  return { outcome: 'completed' }
}
```

```ts
import { openaiApiKey } from './env'
import { getPermit, updatePermit } from './permits'
import { upsertBuildingProfile, type BuildingProfile } from './building-profiles'

export type AnalysisResult = {
  outcome: 'completed' | 'skipped' | 'failed'
  error?: string
}

const SYSTEM_PROMPT = `You are a building systems analyst. Extract structured information from building permit text.

Rules:
- Extract ONLY facts explicitly stated in the permit text
- Use null for any field not mentioned or unclear
- For detectedUpgrades, include only work explicitly described as an upgrade or replacement
- For years in detectedUpgrades, use the year only if explicitly stated; otherwise null
- Do not infer, guess, or hallucinate values
- Confidence scores (0.0–1.0) reflect how clearly the information appears in the text

Respond with valid JSON matching this exact schema:
{
  "buildingYear": number | null,
  "buildingType": string | null,
  "squareFootage": number | null,
  "hvacSystem": string | null,
  "lightingType": string | null,
  "electricalService": string | null,
  "detectedUpgrades": [{ "type": "HVAC"|"Lighting"|"Electrical"|"Envelope"|"Controls"|"Other", "description": string, "year": number | null }],
  "confidenceScores": { [field: string]: number }
}`

function getClient(): OpenAI {
  return new OpenAI({ apiKey: openaiApiKey() })
}

export async function analyzePermit(permitId: string): Promise<AnalysisResult> {
  const permit = await getPermit(permitId)
  if (!permit) throw new Error(`Permit not found: ${permitId}`)

  if (!permit.rawText || permit.textExtraction?.status !== 'completed') {
    return { outcome: 'skipped' }
  }

  const client = getClient()
  const response = await client.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: SYSTEM_PROMPT },
      {
        role: 'user',
        content: `Extract building information from this permit text:\n\n${permit.rawText}`,
      },
    ],
    response_format: { type: 'json_object' },
    temperature: 0,
  })

  const content = response.choices[0].message.content
  if (!content) throw new Error('OpenAI returned empty response')

  const data = JSON.parse(content) as Partial<BuildingProfile>

  const profile: BuildingProfile = {
    permitId,
    buildingYear: data.buildingYear ?? null,
    buildingType: data.buildingType ?? null,
    squareFootage: data.squareFootage ?? null,
    hvacSystem: data.hvacSystem ?? null,
    lightingType: data.lightingType ?? null,
    electricalService: data.electricalService ?? null,
    detectedUpgrades: Array.isArray(data.detectedUpgrades) ? data.detectedUpgrades : [],
    confidenceScores: data.confidenceScores ?? {},
    extractedAt: new Date(),
  }

  await upsertBuildingProfile(profile)
  await updatePermit(permitId, { status: 'analyzed' })

  return { outcome: 'completed' }
}
```

```ts
import { type NextRequest } from 'next/server'
import { extractPermitText } from '@/lib/extraction'
import { analyzePermit } from '@/lib/analysis'
import { computeRetrofitScore } from '@/lib/scoring'
import { getBuildingProfile } from '@/lib/building-profiles'
import { upsertPrediction } from '@/lib/predictions'
import { updatePermit } from '@/lib/permits'

export async function POST(
  _request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params
  try {
    const extractResult = await extractPermitText(id)

    if (extractResult.outcome === 'scanned_pdf') {
      return Response.json({ status: 'scanned_pdf' })
    }

    if (extractResult.outcome === 'failed') {
      return Response.json({ status: 'failed', error: extractResult.error }, { status: 500 })
    }

    const analyzeResult = await analyzePermit(id)

    if (analyzeResult.outcome === 'failed') {
      return Response.json({ status: 'failed', error: analyzeResult.error }, { status: 500 })

    const scored = computeRetrofitScore(profile)
    await upsertPrediction({ permitId: id, generatedAt: new Date(), ...scored })
    await updatePermit(id, { status: 'scored' })

    return Response.json({ status: 'scored' })
  } catch (err) {
    console.error('[process]', err)
    return Response.json(
      { status: 'failed', error: err instanceof Error ? err.message : 'Unknown error' },
      { status: 500 }
    )
  }
}
```

```tsx
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'

interface Props {
  permitId: string
  lastAnalyzedAt: string | null
}

export function AnalysisButton({ permitId, lastAnalyzedAt }: Props) {
  const router = useRouter()
  const [pending, setPending] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const hasRun = lastAnalyzedAt !== null

  async function handleClick() {
    setPending(true)
    setError(null)
    try {
      const res = await fetch(`/api/permits/${permitId}/process`, { method: 'POST' })
      const data = (await res.json()) as { status: string; error?: string }
      if (!res.ok || data.status === 'failed') {
        setError(data.error ?? 'Analysis failed. Please try again.')
      } else {
        router.refresh()
      }
    } catch {
      setError('Network error. Please try again.')
    } finally {
      setPending(false)
    }
  }

  return (
    <div className="space-y-2">
      {hasRun && (
        <p className="text-xs text-gray-500">
          Last analyzed: {new Date(lastAnalyzedAt).toLocaleString()}
        </p>
      )}
      <button
        onClick={handleClick}
        disabled={pending}
        className="rounded-md bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
      >
        {pending ? 'Analyzing…' : hasRun ? 'Re-Run Analysis' : 'Run AI Retrofit Analysis'}
      </button>
      {error && <p className="mt-1 text-sm text-red-600 dark:text-red-400">{error}</p>}
    </div>
  )
}
```
